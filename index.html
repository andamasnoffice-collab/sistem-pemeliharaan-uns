Integrasi Auto-Write Google Spreadsheet

Setiap data bertambah/diubah otomatis langsung tertulis ke Google Sheets

Tempelkan URL Web App dari Apps Script Google Sheets Anda di sini.

Auto-Write Realtime (Tanpa Klik Button):

Jika aktif, setiap penambahan/pengubahan kilometer, transaksi servis, atau armada akan langsung mengirimkan update ke Google Sheets tanpa perlu konfirmasi lagi.

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  if (data.type === 'SYNC_ALL' || data.type === 'AUTO_WRITE') {
    var sheetV = ss.getSheetByName('Master Armada') || ss.insertSheet('Master Armada');
    sheetV.clear();
    sheetV.appendRow(['Plat Nomor', 'Model Armada', 'Sopir', 'Kategori', 'Odometer Terkini (KM)', 'Odometer Servis Terakhir (KM)', 'Status']);
    if (data.vehicles && data.vehicles.length) {
      data.vehicles.forEach(function(v) {
        sheetV.appendRow([v.plate, v.model, v.driver || '-', v.category || 'Operasional', v.currentKm || 0, v.lastServiceKm || 0, v.isUnderService ? 'Dalam Pengajuan Servis' : 'Aktif']);
      });
    }
    
    var sheetL = ss.getSheetByName('Riwayat Servis') || ss.insertSheet('Riwayat Servis');
    sheetL.clear();
    sheetL.appendRow(['ID Log', 'Tanggal', 'Plat Nomor', 'Bengkel', 'No Invoice', 'Odometer (KM)', 'Total Biaya (Rp)']);
    if (data.logs && data.logs.length) {
      data.logs.forEach(function(l) {
        sheetL.appendRow([l.id, l.date, l.vehicleId, l.workshop, l.invoiceNo || '-', l.km || 0, l.totalCost || 0]);
      });
    }
  }
  return ContentService.createTextOutput(JSON.stringify({status: 'success'})).setMimeType(ContentService.MimeType.JSON);
}
